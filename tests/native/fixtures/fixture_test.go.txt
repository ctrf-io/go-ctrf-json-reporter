package fixture

import (
 "os"
 "testing"
 "time"
)

func TestPass(t *testing.T) {time.Sleep(12*time.Millisecond)}

func TestParallelA(t *testing.T) {t.Parallel();time.Sleep(25*time.Millisecond)}
func TestParallelB(t *testing.T) {t.Parallel();time.Sleep(25*time.Millisecond)}

 time.Sleep(17*time.Millisecond)
 if os.Getenv("CTRF_FIXTURE_OUTCOME")=="failed" {t.Fatal("independent retry failure")}
}
